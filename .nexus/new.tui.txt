class AIProviderClient:
    """Universal AI provider client with latest APIs"""
    
    def __init__(self, config: NexusCodeConfig):
        self.config = config
        self.clients = {}
        self._initialize_clients()
        
    def _initialize_clients(self):
        """Initialize all AI provider clients"""
        
        # OpenAI with Responses API
        if self.config.api_keys["OPENAI_API_KEY"]:
            self.clients[AIProvider.OPENAI] = openai.OpenAI(
                api_key=self.config.api_keys["OPENAI_API_KEY"]
            )
            
        # Anthropic with Messages API
        if self.config.api_keys["ANTHROPIC_API_KEY"]:
            self.clients[AIProvider.ANTHROPIC] = anthropic.Anthropic(
                api_key=self.config.api_keys["ANTHROPIC_API_KEY"]
            )
            
        # Gemini with latest API
        if self.config.api_keys["GEMINI_API_KEY"]:
            gemini_ai.configure(api_key=self.config.api_keys["GEMINI_API_KEY"])
            self.clients[AIProvider.GEMINI] = gemini_ai
            
        # Ollama local client
        try:
            self.clients[AIProvider.OLLAMA] = ollama.Client()
        except Exception as e:
            logger.warning(f"Could not connect to Ollama: {e}")
    
    async def generate_response(self, model: AIModel, messages: List[Dict], **kwargs) -> Dict:
        """Generate response using the appropriate AI provider with latest APIs"""
        
        if model.provider == AIProvider.OPENAI:
            return await self._openai_responses_api(model, messages, **kwargs)
        elif model.provider == AIProvider.ANTHROPIC:
            return await self._anthropic_messages_api(model, messages, **kwargs)
        elif model.provider == AIProvider.GEMINI:
            return await self._gemini_api(model, messages, **kwargs)
        elif model.provider == AIProvider.OLLAMA:
            return await self._ollama_api(model, messages, **kwargs)
        else:
            raise ValueError(f"Unsupported provider: {model.provider}")
    
    async def _openai_responses_api(self, model: AIModel, messages: List[Dict], **kwargs) -> Dict:
        """Use OpenAI's new Responses API"""
        client = self.clients[AIProvider.OPENAI]
        
        try:
            # Convert messages to input format for Responses API
            input_content = messages[-1]["content"] if messages else ""
            
            # Previous conversation context
            previous_response_id = kwargs.get("previous_response_id")
            
            response = client.responses.create(
                model=model.name,
                input=input_content,
                previous_response_id=previous_response_id,
                max_tokens=kwargs.get("max_tokens", model.max_tokens),
                temperature=kwargs.get("temperature", 0.7)
            )
            
            return {
                "id": response.id,
                "content": response.output[0].content[0].text if response.output else "",
                "usage": getattr(response, "usage", {}),
                "provider": "openai"
            }
            
        except Exception as e:
            logger.error(f"OpenAI Responses API error: {e}")
            raise
    
    async def _anthropic_messages_api(self, model: AIModel, messages: List[Dict], **kwargs) -> Dict:
        """Use Anthropic's Messages API with latest beta features"""
        client = self.clients[AIProvider.ANTHROPIC]
        
        try:
            # Configure beta headers
            extra_headers = {}
            for beta_feature in model.beta_features:
                extra_headers["anthropic-beta"] = beta_feature
                
            # Add context management if supported
            if "context-management-2025-06-27" in model.beta_features:
                extra_headers["anthropic-beta"] += ",context-management-2025-06-27"
            
            # Configure caching if large context
            system_content = kwargs.get("system", "")
            if len(system_content) > 10000:  # Enable caching for large contexts
                system = [
                    {"type": "text", "text": system_content, "cache_control": {"type": "ephemeral"}}
                ]
            else:
                system = system_content
            
            response = client.messages.create(
                model=model.name,
                max_tokens=kwargs.get("max_tokens", model.max_tokens),
                temperature=kwargs.get("temperature", 0.7),
                system=system,
                messages=messages,
                extra_headers=extra_headers
            )
            return {
                "id": response.id,
                "content": response.content[0].text,
                "usage": response.usage.__dict__,
                "provider": "anthropic"
            }
            
        except Exception as e:
            logger.error(f"Anthropic Messages API error: {e}")
            raise
    
    async def _gemini_api(self, model: AIModel, messages: List[Dict], **kwargs) -> Dict:
        """Use Google's Gemini API with latest features"""
        
        try:
            # Configure model
            generation_config = {
                "temperature": kwargs.get("temperature", 0.7),
                "max_output_tokens": kwargs.get("max_tokens", model.max_tokens),
            }
            
            # Initialize model
            gemini_model = gemini_ai.GenerativeModel(
                model_name=model.name,
                generation_config=generation_config
            )
            
            # Convert messages to Gemini format
            gemini_messages = []
            for message in messages:
                if message["role"] == "user":
                    gemini_messages.append(message["content"])
                elif message["role"] == "assistant":
                    # Add to conversation history
                    pass
            
            # Generate response
            response = gemini_model.generate_content(gemini_messages[-1])
            
            return {
                "id": getattr(response, "id", "gemini-" + str(int(time.time()))),
                "content": response.text,
                "usage": {"input_tokens": 0, "output_tokens": 0},  # Gemini doesn't provide detailed usage
                "provider": "gemini"
            }
            
        except Exception as e:
            logger.error(f"Gemini API error: {e}")
            raise
    
    async def _ollama_api(self, model: AIModel, messages: List[Dict], **kwargs) -> Dict:
        """Use Ollama for local models"""
        client = self.clients[AIProvider.OLLAMA]
        
        try:
            response = client.chat(
                model=model.name,
                messages=messages,
                stream=False,
                options={
                    "temperature": kwargs.get("temperature", 0.7),
                    "num_ctx": model.context_length,
                    "num_predict": kwargs.get("max_tokens", model.max_tokens)
                }
            )
            
            return {
                "id": f"ollama-{int(time.time())}",
                "content": response["message"]["content"],
                "usage": {"input_tokens": 0, "output_tokens": 0},
                "provider": "ollama"
            }
            
        except Exception as e:
            logger.error(f"Ollama API error: {e}")
            raise


class MCPManager:
    """Model Context Protocol integration manager"""
    
    def __init__(self, config: NexusCodeConfig):
        self.config = config
        self.servers = {}
        self.sessions = {}
        self.tools = {}
        
    async def initialize(self):
        """Initialize all MCP servers"""
        logger.info("Initializing MCP servers...")
        
        for name, server_config in self.config.mcp_servers.items():
            if server_config.enabled:
                try:
                    await self._start_mcp_server(name, server_config)
                    logger.info(f"MCP server '{name}' initialized successfully")
                except Exception as e:
                    logger.error(f"Failed to initialize MCP server '{name}': {e}")
    
    async def _start_mcp_server(self, name: str, config: MCPServer):
        """Start an individual MCP server"""
        
        if config.transport == "stdio":
            server_params = StdioServerParameters(
                command=config.command,
                args=config.args,
                env=None
            )
        elif config.transport == "http":
            server_params = SSEServerParameters(
                url=config.url
            )
        else:
            raise ValueError(f"Unsupported transport: {config.transport}")
        
        # Create session
        session = ClientSession()
        self.sessions[name] = session
        
        # Connect to server
        await session.initialize()
        
        # Discover tools if auto-discovery is enabled
        if config.auto_discover:
            tools = await self._discover_tools(session)
            self.tools[name] = tools
            config.tools = list(tools.keys())
    
    async def _discover_tools(self, session: ClientSession) -> Dict:
        """Discover tools from an MCP server"""
        try:
            tools_response = await session.list_tools()
            tools = {}
            
            for tool in tools_response.tools:
                tools[tool.name] = {
                    "description": tool.description,
                    "input_schema": tool.inputSchema,
                    "output_schema": getattr(tool, "outputSchema", None)
                }
            
            return tools
        except Exception as e:
            logger.error(f"Failed to discover tools: {e}")
            return {}
    
    async def execute_tool(self, server_name: str, tool_name: str, arguments: Dict) -> Any:
        """Execute a tool on an MCP server"""
        
        if server_name not in self.sessions:
            raise ValueError(f"MCP server '{server_name}' not found")
        
        session = self.sessions[server_name]
        
        try:
            result = await session.call_tool(tool_name, arguments)
            return result
        except Exception as e:
            logger.error(f"Failed to execute tool '{tool_name}' on server '{server_name}': {e}")
            raise
    
    async def shutdown(self):
        """Shutdown all MCP servers"""
        for name, session in self.sessions.items():
            try:
                await session.close()
                logger.info(f"MCP server '{name}' shutdown successfully")
            except Exception as e:
                logger.error(f"Failed to shutdown MCP server '{name}': {e}")


class AgentOrchestrator:
    """Intelligent agent orchestration system"""
    
    def __init__(self, config: NexusCodeConfig, ai_client: AIProviderClient, mcp_manager: MCPManager):
        self.config = config
        self.ai_client = ai_client
        self.mcp_manager = mcp_manager
        self.agents = config.agents
        self.active_conversations = {}
        self.task_queue = asyncio.Queue()
        self.results = {}
        
    async def execute_task(self, task: Task) -> Dict:
        """Execute a task using appropriate agents"""
        logger.info(f"Executing task: {task.id}")
        
        # Determine optimal agent for task
        agent = self._select_agent(task)
        
        # Create conversation context
        conversation_id = f"{task.id}_{agent.id}_{int(time.time())}"
        self.active_conversations[conversation_id] = {
            "task": task,
            "agent": agent,
            "messages": [],
            "tools_used": [],
            "iterations": 0
        }
        
        try:
            result = await self._run_agent_loop(conversation_id)
            self.results[task.id] = result
            
            # Check for handoff conditions
            if self._should_handoff(agent, result):
                next_agent = self._get_handoff_agent(agent, task)
                if next_agent:
                    logger.info(f"Handing off from {agent.id} to {next_agent.id}")
                    return await self._handoff_task(task, next_agent, result)
            
            return result
            
        except Exception as e:
            logger.error(f"Failed to execute task {task.id}: {e}")
            raise
    
    def _select_agent(self, task: Task) -> Agent:
        """Select the best agent for a task based on requirements"""
        
        # Simple heuristic-based selection
        if "architecture" in task.description.lower() or "design" in task.description.lower():
            return self.agents["architect"]
        elif "code" in task.description.lower() or "implement" in task.description.lower():
            return self.agents["developer"] 
        elif "review" in task.description.lower() or "check" in task.description.lower():
            return self.agents["reviewer"]
        else:
            # Default to architect for planning
            return self.agents["architect"]
    
    async def _run_agent_loop(self, conversation_id: str) -> Dict:
        """Run the main agent execution loop"""
        conv = self.active_conversations[conversation_id]
        agent = conv["agent"]
        task = conv["task"]
        
        # Initial system message
        messages = [{
            "role": "system",
            "content": f"{agent.system_prompt}\n\nTask: {task.description}"
        }]
        
        # Add task context
        messages.append({
            "role": "user", 
            "content": f"Execute this task: {task.description}\n\nPriority: {task.priority.value}"
        })
        
        max_iterations = agent.max_iterations
        results = []
        
        for iteration in range(max_iterations):
            conv["iterations"] = iteration + 1
            
            try:
                # Generate response
                response = await self.ai_client.generate_response(
                    model=agent.model,
                    messages=messages,
                    temperature=agent.temperature
                )
                
                messages.append({
                    "role": "assistant",
                    "content": response["content"]
                })
                
                # Check for tool calls (if supported)
                if self._has_tool_calls(response):
                    tool_results = await self._execute_tool_calls(agent, response)
                    conv["tools_used"].extend(tool_results)
                    
                    # Add tool results to conversation
                    for tool_result in tool_results:
                        messages.append({
                            "role": "user",
                            "content": f"Tool result: {tool_result}"
                        })
                
                results.append(response)
                
                # Check if task is complete
                if self._is_task_complete(response["content"]):
                    break
                    
            except Exception as e:
                logger.error(f"Agent loop error on iteration {iteration}: {e}")
                break
        
        return {
            "conversation_id": conversation_id,
            "agent_id": agent.id,
            "task_id": task.id,
            "iterations": conv["iterations"],
            "messages": messages,
            "tools_used": conv["tools_used"],
            "final_result": results[-1] if results else None,
            "status": "completed" if results else "failed"
        }
    
    def _has_tool_calls(self, response: Dict) -> bool:
        """Check if response contains tool calls"""
        content = response.get("content", "")
        
        # Simple heuristic - look for tool call patterns
        tool_patterns = [
            "```tool",
            "TOOL:",
            "execute_tool",
            "call_function"
        ]
        
        return any(pattern in content for pattern in tool_patterns)
    
    async def _execute_tool_calls(self, agent: Agent, response: Dict) -> List[Dict]:
        """Execute tool calls for an agent"""
        tools_used = []
        
        # Parse tool calls from response (simplified)
        content = response.get("content", "")
        
        # For demo, we'll execute filesystem tools
        if "filesystem" in agent.tools:
            if "create_file" in content.lower():
                try:
                    result = await self.mcp_manager.execute_tool(
                        "filesystem",
                        "write_file",
                        {"path": "/tmp/test.txt", "content": "Generated by NEXUS CODE"}
                    )
                    tools_used.append({
                        "tool": "filesystem.write_file",
                        "result": result
                    })
                except Exception as e:
                    logger.error(f"Tool execution failed: {e}")
        
        return tools_used
    
    def _is_task_complete(self, content: str) -> bool:
        """Determine if a task is complete based on agent response"""
        completion_indicators = [
            "task completed",
            "implementation finished", 
            "work done",
            "completed successfully",
            "task finished"
        ]
        
        return any(indicator in content.lower() for indicator in completion_indicators)
    
    def _should_handoff(self, agent: Agent, result: Dict) -> bool:
        """Determine if task should be handed off to another agent"""
        
        # Check handoff conditions
        for condition in agent.handoff_conditions:
            if condition in result.get("final_result", {}).get("content", ""):
                return True
        
        return False
    
    def _get_handoff_agent(self, current_agent: Agent, task: Task) -> Optional[Agent]:
        """Get the next agent for handoff"""
        
        if current_agent.role == AgentRole.ARCHITECT:
            return self.agents.get("developer")
        elif current_agent.role == AgentRole.DEVELOPER:
            return self.agents.get("reviewer")
        
        return None
    
    async def _handoff_task(self, task: Task, next_agent: Agent, previous_result: Dict) -> Dict:
        """Hand off task to next agent"""
        
        # Create handoff task
        handoff_task = Task(
            id=f"{task.id}_handoff",
            description=f"Continue work from {previous_result['agent_id']}: {task.description}",
            priority=task.priority,
            dependencies=[task.id]
        )
        
        return await self.execute_task(handoff_task)

    async def create_project(self, template_name: str, project_name: str, target_dir: Path) -> Project:
        """Create a new project from template"""
        
        if template_name not in self.templates:
            raise ValueError(f"Template '{template_name}' not found")
        
        template = self.templates[template_name]
        project_dir = target_dir / project_name
        project_dir.mkdir(exist_ok=True)
        
        # Create files from template
        for file_path, content in template["files"].items():
            if content is None:  # Directory
                (project_dir / file_path).mkdir(parents=True, exist_ok=True)
            else:
                file_location = project_dir / file_path
                file_location.parent.mkdir(parents=True, exist_ok=True)
                
                with open(file_location, "w") as f:
                    f.write(content)
        
        # Create project object
        project = Project(
            name=project_name,
            description=template["description"],
            template=template_name,
            languages=self._detect_languages(template),
            frameworks=self._detect_frameworks(template),
            files=list(template["files"].keys())
        )
        
        self.projects[project_name] = project
        
        logger.info(f"Project '{project_name}' created from template '{template_name}'")
        return project
    
    def _detect_languages(self, template: Dict) -> List[str]:
        """Detect programming languages in template"""
        languages = set()
        
        for file_path in template["files"].keys():
            if file_path.endswith(".py"):
                languages.add("Python")
            elif file_path.endswith((".js", ".jsx", ".ts", ".tsx")):
                languages.add("JavaScript/TypeScript")
            elif file_path.endswith(".go"):
                languages.add("Go")
            elif file_path.endswith((".cpp", ".c", ".h")):
                languages.add("C/C++")
        
        return list(languages)
    
    def _detect_frameworks(self, template: Dict) -> List[str]:
        """Detect frameworks in template"""
        frameworks = set()
        
        for file_path, content in template["files"].items():
            if content and "react" in content.lower():
                frameworks.add("React")
            elif content and "fastapi" in content.lower():
                frameworks.add("FastAPI")
            elif content and "django" in content.lower():
                frameworks.add("Django")
        
        return list(frameworks)
