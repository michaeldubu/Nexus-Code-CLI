@dataclass
class ImageGenerationRequest:
    """Image generation request configuration"""
    prompt: str
    negative_prompt: Optional[str] = None
    model: ImageModel = ImageModel.DALLE_3
    style: ImageStyle = ImageStyle.PHOTOREALISTIC
    quality: ImageQuality = ImageQuality.HIGH
    width: int = 1024
    height: int = 1024
    steps: int = 50
    cfg_scale: float = 7.5
    seed: Optional[int] = None
    batch_size: int = 1
    variations: int = 1
    upscale_factor: float = 1.0
    enhance: bool = False
    custom_params: Dict[str, Any] = field(default_factory=dict)


@dataclass
class ImageGenerationResult:
    """Image generation result"""
    id: str
    request: ImageGenerationRequest
    images: List[str]  # Base64 encoded or URLs
    metadata: Dict[str, Any]
    generation_time: float
    cost: float = 0.0
    provider: str = "unknown"

    async def generate_image(self, request: ImageGenerationRequest) -> ImageGenerationResult:
        """Generate a single image using the revolutionary system"""
        
        logger.info(f"🎨 Generating image: {request.prompt[:50]}...")
        start_time = time.time()
        
        try:
            if request.model in [ImageModel.DALLE_3, ImageModel.DALLE_2]:
                result = await self._generate_with_openai(request)
            elif request.model.value.startswith("flux"):
                result = await self._generate_with_flux(request)
            elif request.model == ImageModel.STABLE_DIFFUSION_XL:
                result = await self._generate_with_stability(request)
            else:
                raise ValueError(f"Unsupported model: {request.model}")
            
            generation_time = time.time() - start_time
            result.generation_time = generation_time
            
            # Cache result for optimization
            cache_key = self._get_cache_key(request)
            self.image_cache[cache_key] = result
            
            logger.info(f"✅ Image generated in {generation_time:.2f}s")
            return result

        except Exception as e:
            logger.error(f"❌ Image generation failed: {e}")
            raise
    
    async def _generate_with_openai(self, request: ImageGenerationRequest) -> ImageGenerationResult:
        """Generate image using OpenAI GPT-image-1.5"""
        
        if not self.openai_client:
            raise ValueError("OpenAI client not initialized")
        
        try:
            # Enhanced prompt engineering for DALL-E 3
            enhanced_prompt = self._enhance_prompt_for_dalle(request)
            
            response = self.openai_client.images.generate(
                model=request.model.value,
                prompt=enhanced_prompt,
                size=f"{request.width}x{request.height}",
                quality="hd" if request.quality in [ImageQuality.HIGH, ImageQuality.ULTRA] else "standard",
                style="vivid" if request.style == ImageStyle.ARTISTIC else "natural",
                n=min(request.variations, 4),  # DALL-E 3 max is 4
                response_format="url"
            )
            
            images = [img.url for img in response.data]
            
            # Download and convert to base64 for consistent handling
            base64_images = []
            for url in images:
                async with aiohttp.ClientSession() as session:
                    async with session.get(url) as resp:
                        image_data = await resp.read()
                        base64_img = base64.b64encode(image_data).decode('utf-8')
                        base64_images.append(base64_img)
            
            return ImageGenerationResult(
                id=f"dalle_{int(time.time())}",
                request=request,
                images=base64_images,
                metadata={
                    "model": request.model.value,
                    "revised_prompt": getattr(response.data[0], 'revised_prompt', enhanced_prompt),
                    "original_prompt": request.prompt,
                    "size": f"{request.width}x{request.height}"
                },
                generation_time=0.0,  # Will be set by caller
                provider="openai"
            )
            
        except Exception as e:
            logger.error(f"OpenAI generation error: {e}")
            raise
    
    async def _generate_with_flux(self, request: ImageGenerationRequest) -> ImageGenerationResult:
        """Generate image using Flux models via Replicate"""
        
        if not self.replicate_client:
            raise ValueError("Replicate client not initialized")
        
        try:
            # Select Flux model variant
            if request.model == ImageModel.FLUX_SCHNELL:
                model = "black-forest-labs/flux-schnell"
            elif request.model == ImageModel.FLUX_PRO:
                model = "black-forest-labs/flux-1.1-pro"
            elif request.model == ImageModel.FLUX_DEV:
                model = "black-forest-labs/flux-dev"
            else:
                model = "black-forest-labs/flux-schnell"
            
            # Enhanced prompt for Flux
            enhanced_prompt = self._enhance_prompt_for_flux(request)
            
            # Run Flux generation
            output = replicate.run(
                model,
                input={
                    "prompt": enhanced_prompt,
                    "width": request.width,
                    "height": request.height,
                    "num_inference_steps": request.steps,
                    "guidance_scale": request.cfg_scale,
                    "seed": request.seed,
                    "num_outputs": request.variations,
                    "output_format": "png",
                    "output_quality": 95 if request.quality == ImageQuality.ULTRA else 85
                }
            )
            
            # Handle Flux output (can be single URL or list)
            if isinstance(output, str):
                image_urls = [output]
            elif isinstance(output, list):
                image_urls = output
            else:
                raise ValueError(f"Unexpected Flux output format: {type(output)}")
            
            # Download and convert to base64
            base64_images = []
            for url in image_urls:
                async with aiohttp.ClientSession() as session:
                    async with session.get(url) as resp:
                        image_data = await resp.read()
                        base64_img = base64.b64encode(image_data).decode('utf-8')
                        base64_images.append(base64_img)
            
            return ImageGenerationResult(
                id=f"flux_{int(time.time())}",
                request=request,
                images=base64_images,
                metadata={
                    "model": model,
                    "prompt": enhanced_prompt,
                    "steps": request.steps,
                    "guidance_scale": request.cfg_scale,
                    "seed": request.seed
                },
                generation_time=0.0,
                provider="replicate"
            )
            
        except Exception as e:
            logger.error(f"Flux generation error: {e}")
            raise
    
    async def _generate_with_stability(self, request: ImageGenerationRequest) -> ImageGenerationResult:
        """Generate image using Stability AI"""
        
        if not self.stability_client:
            raise ValueError("Stability AI client not initialized")
        
        try:
            from stability_sdk import generation
            
            # Enhanced prompt for Stable Diffusion
            enhanced_prompt = self._enhance_prompt_for_sdxl(request)
            
            # Generate image
            answers = self.stability_client.generate(
                prompt=enhanced_prompt,
                negative_prompt=request.negative_prompt or "",
                width=request.width,
                height=request.height,
                cfg_scale=request.cfg_scale,
                steps=request.steps,
                seed=request.seed or 0,
                samples=request.variations,
                engine="stable-diffusion-xl-1024-v1-0"
            )
            
            base64_images = []
            for answer in answers:
                for artifact in answer.artifacts:
                    if artifact.finish_reason == generation.FILTER:
                        logger.warning("Image filtered by safety classifier")
                        continue
                    if artifact.type == generation.ARTIFACT_IMAGE:
                        base64_img = base64.b64encode(artifact.binary).decode('utf-8')
                        base64_images.append(base64_img)
            
            return ImageGenerationResult(
                id=f"sdxl_{int(time.time())}",
                request=request,
                images=base64_images,
                metadata={
                    "model": "stable-diffusion-xl",
                    "prompt": enhanced_prompt,
                    "negative_prompt": request.negative_prompt,
                    "steps": request.steps,
                    "cfg_scale": request.cfg_scale
                },
                generation_time=0.0,
                provider="stability"
            )
            
        except Exception as e:
            logger.error(f"Stability AI generation error: {e}")
            raise
    
    def _enhance_prompt_for_dalle(self, request: ImageGenerationRequest) -> str:
        """Enhance prompt specifically for DALL-E 3"""
        
        prompt = request.prompt
        
        # Add style modifiers
        if request.style == ImageStyle.PHOTOREALISTIC:
            prompt += ", photorealistic, high detail, professional photography"
        elif request.style == ImageStyle.ARTISTIC:
            prompt += ", artistic style, creative interpretation, beautiful composition"
        elif request.style == ImageStyle.CYBERPUNK:
            prompt += ", cyberpunk aesthetic, neon lights, futuristic cityscape, digital art"
        elif request.style == ImageStyle.REVOLUTIONARY:
            prompt += ", revolutionary design, cutting-edge technology, innovative concept, futuristic"
        
        # Add quality modifiers
        if request.quality == ImageQuality.ULTRA:
            prompt += ", ultra high quality, masterpiece, award-winning"
        elif request.quality == ImageQuality.PROFESSIONAL:
            prompt += ", professional quality, studio lighting, perfect composition"
        
        return prompt
    
    def _enhance_prompt_for_flux(self, request: ImageGenerationRequest) -> str:
        """Enhance prompt specifically for Flux models"""
        
        prompt = request.prompt
        
        # Flux-specific enhancements
        style_mappings = {
            ImageStyle.PHOTOREALISTIC: "RAW photo, 8k uhd, dslr, high quality, film grain, Fujifilm XT3",
            ImageStyle.ARTISTIC: "concept art, digital painting, trending on artstation, highly detailed",
            ImageStyle.CYBERPUNK: "cyberpunk style, neon, dark, futuristic, blade runner, ghost in the shell",

        if request.style in style_mappings:

        # Add quality enhancement

            prompt += ", masterpiece, best quality, ultra-detailed"

    def _enhance_prompt_for_sdxl(self, request: ImageGenerationRequest) -> str:

        # SDXL works well with detailed, descriptive prompts

            prompt += ", (photorealistic:1.4), detailed, sharp focus, professional photography"
        elif request.style == ImageStyle.ARTISTIC:
            prompt += ", (digital art:1.3), concept art, illustration, detailed"

    def _get_cache_key(self, request: ImageGenerationRequest) -> str:
        """Generate cache key for image request"""
        
        key_data = {
            "prompt": request.prompt,
            "model": request.model.value,
            "style": request.style.value,
            "width": request.width,
            "height": request.height,
            "seed": request.seed
        }
        
        key_string = json.dumps(key_data, sort_keys=True)
        return hashlib.md5(key_string.encode()).hexdigest()
    
    async def generate_batch(self, requests: List[ImageGenerationRequest]) -> BatchImageJob:
        """Generate multiple images in batch for optimal performance"""
        
        job_id = f"batch_{int(time.time())}"
        
        batch_job = BatchImageJob(
            id=job_id,
            requests=requests,
            status="processing"
        )
        
        self.batch_jobs[job_id] = batch_job
        logger.info(f"🎨 Starting batch generation job {job_id} with {len(requests)} requests")
        
        try:
            results = []
            total_requests = len(requests)
            
            # Process requests concurrently with rate limiting
            semaphore = asyncio.Semaphore(3)  # Max 3 concurrent generations
            
            async def process_request(req_index, request):
                async with semaphore:
                    try:
                        result = await self.generate_image(request)
                        
                        # Update progress
                        batch_job.progress = (req_index + 1) / total_requests * 100
                        logger.info(f"Batch progress: {batch_job.progress:.1f}%")
                        
                        return result
                    except Exception as e:
                        logger.error(f"Batch request {req_index} failed: {e}")
                        return None
            
            # Execute all requests concurrently
            tasks = [process_request(i, req) for i, req in enumerate(requests)]
            results = await asyncio.gather(*tasks, return_exceptions=True)
            
            # Filter successful results
            successful_results = [r for r in results if r is not None and not isinstance(r, Exception)]
            
            batch_job.results = successful_results
            batch_job.status = "completed"
            batch_job.completed_at = time.time()
            batch_job.progress = 100.0
            
            logger.info(f"✅ Batch job {job_id} completed: {len(successful_results)}/{total_requests} successful")
            
            return batch_job
            
        except Exception as e:
            batch_job.status = "failed"
            batch_job.error_message = str(e)
            logger.error(f"❌ Batch job {job_id} failed: {e}")
            raise
    
    async def enhance_image(self, image_data: str, enhancement_type: str = "upscale") -> str:
        """Enhance existing images with AI-powered improvements"""
        
        try:
            # Decode base64 image
            image_bytes = base64.b64decode(image_data)
            image = Image.open(io.BytesIO(image_bytes))
            
            if enhancement_type == "upscale":
                # Use Real-ESRGAN for upscaling via Replicate
                if self.replicate_client:
                    # Convert image to temporary URL or base64
                    buffered = io.BytesIO()
                    image.save(buffered, format="PNG")
                    img_base64 = base64.b64encode(buffered.getvalue()).decode()
                    
                    output = replicate.run(
                        "nightmareai/real-esrgan:42fed1c4974146d4d2414e2be2c5277c7fcf05fcc972f6b7c4c8da",
                        input={
                            "image": f"data:image/png;base64,{img_base64}",
                            "scale": 4,
                            "face_enhance": True
                        }
                    )
                    
                    # Download enhanced image
                    async with aiohttp.ClientSession() as session:
                        async with session.get(output) as resp:
                            enhanced_data = await resp.read()
                            return base64.b64encode(enhanced_data).decode()
            
            elif enhancement_type == "denoise":
                # Apply denoising filters
                enhanced = image.filter(ImageFilter.MedianFilter(size=3))
                buffered = io.BytesIO()
                enhanced.save(buffered, format="PNG")
                return base64.b64encode(buffered.getvalue()).decode()
            
            elif enhancement_type == "sharpen":
                # Apply sharpening
                enhancer = ImageEnhance.Sharpness(image)
                enhanced = enhancer.enhance(1.5)
                buffered = io.BytesIO()
                enhanced.save(buffered, format="PNG")
                return base64.b64encode(buffered.getvalue()).decode()
            
            elif enhancement_type == "color_enhance":
                # Enhance colors and contrast
                color_enhancer = ImageEnhance.Color(image)
                enhanced = color_enhancer.enhance(1.2)
                
                contrast_enhancer = ImageEnhance.Contrast(enhanced)
                enhanced = contrast_enhancer.enhance(1.1)
                
                buffered = io.BytesIO()
                enhanced.save(buffered, format="PNG")
                return base64.b64encode(buffered.getvalue()).decode()
            
            else:
                raise ValueError(f"Unsupported enhancement type: {enhancement_type}")
                
        except Exception as e:
            logger.error(f"Image enhancement failed: {e}")
            raise

        logger.info(f"🎭 Starting multimodal collaboration: {task_description}")
        
        try:
            # Step 1: Text analysis and planning
            planning_prompt = f"""
            Analyze this creative task and create a comprehensive plan:
            
            Task: {task_description}
            
            Provide:
            1. Visual concept description for image generation
            2. Key visual elements and style requirements  
            3. Color palette and mood
            4. Any animation or video requirements
            5. Text content and messaging
            
            Format as JSON with keys: visual_concept, style_requirements, color_palette, mood, animation_needs, text_content
            """
            
            # Use Claude for sophisticated planning
            claude_model = self.config.models["claude-sonnet-4"]
            planning_response = await self.ai_client.generate_response(
                model=claude_model,
                messages=[{"role": "user", "content": planning_prompt}]
            )
            
            # Parse planning response
            planning_data = self._extract_json_from_response(planning_response["content"])
            
            # Step 2: Generate primary visual asset
            image_request = ImageGenerationRequest(
                prompt=planning_data.get("visual_concept", task_description),
                model=ImageModel.FLUX_PRO,
                style=self._map_mood_to_style(planning_data.get("mood", "professional")),
                quality=ImageQuality.ULTRA,
                width=1024,
                height=1024
            )
            
            primary_image = await self.image_generator.generate_image(image_request)
            
            # Step 3: Generate variations if needed
            variations = []
            if planning_data.get("animation_needs"):
                # Generate video
                video_url = await self.video_generator.generate_video(
                    prompt=planning_data.get("visual_concept", task_description),
                    image_input=f"data:image/png;base64,{primary_image.images[0]}"
                )
                variations.append({"type": "video", "url": video_url})
            
            # Step 4: Generate supporting visuals
            if len(planning_data.get("style_requirements", [])) > 1:
                for i, style_req in enumerate(planning_data["style_requirements"][:3]):
                    variation_request = ImageGenerationRequest(
                        prompt=f"{planning_data.get('visual_concept')} with {style_req}",
                        model=ImageModel.DALLE_3,
                        style=ImageStyle.ARTISTIC,
                        width=512,
                        height=512
                    )
                    
                    variation_result = await self.image_generator.generate_image(variation_request)
                    variations.append({
                        "type": "image_variation",
                        "style": style_req,
                        "data": variation_result.images[0]
                    })
            
            # Step 5: Generate final presentation text
            presentation_prompt = f"""
            Create compelling presentation text for this multimodal project:
            
            Original task: {task_description}
            Visual concept: {planning_data.get('visual_concept')}
            Mood: {planning_data.get('mood')}
            
            Provide:
            1. Project title
            2. Executive summary (2-3 sentences)
            3. Key features list
            4. Technical specifications
            5. Revolutionary impact statement
            
            Use SAAAM Intelligence branding and emphasize innovation.
            """

            presentation_response = await self.ai_client.generate_response(
                model=claude_model,
                messages=[{"role": "user", "content": presentation_prompt}]
            )
            
            return {
                "task_description": task_description,
                "planning_data": planning_data,
                "primary_image": {
                    "data": primary_image.images[0],
                    "metadata": primary_image.metadata
                },
                "variations": variations,
                "presentation_text": presentation_response["content"],
                "generation_time": primary_image.generation_time,
                "collaboration_summary": {
                    "models_used": [claude_model.name, primary_image.metadata["model"]],
                    "total_assets": 1 + len(variations),
                    "innovative_features": ["multimodal_planning", "ai_orchestration", "revolutionary_design"]
                }
            }
            
        except Exception as e:
            logger.error(f"Multimodal collaboration failed: {e}")
            raise
    
    def _extract_json_from_response(self, response_text: str) -> Dict[str, Any]:
        """Extract JSON from AI response"""
        try:
            # Try to find JSON in the response
            import re
            json_pattern = r'\{.*\}'
            json_match = re.search(json_pattern, response_text, re.DOTALL)
            
            if json_match:
                return json.loads(json_match.group())
            else:
                # Fallback to default structure
                return {
                    "visual_concept": response_text[:200],
                    "style_requirements": ["modern", "professional"],
                    "color_palette": ["blue", "white", "black"],
                    "mood": "innovative",
                    "animation_needs": False,
                    "text_content": "Revolutionary AI-powered solution"
                }
                
        except Exception as e:
            logger.error(f"JSON extraction failed: {e}")
            return {"visual_concept": response_text[:200]}
    
    def _map_mood_to_style(self, mood: str) -> ImageStyle:
        """Map mood to appropriate image style"""
        mood_mapping = {
            "innovative": ImageStyle.FUTURISTIC,
            "professional": ImageStyle.MINIMALIST,
            "creative": ImageStyle.ARTISTIC,
            "revolutionary": ImageStyle.REVOLUTIONARY,
            "cyberpunk": ImageStyle.CYBERPUNK,
            "vintage": ImageStyle.VINTAGE
        }
        
        return mood_mapping.get(mood.lower(), ImageStyle.PHOTOREALISTIC)


class ImageWorkflowOrchestrator:
    """Revolutionary workflow orchestrator for complex image generation tasks"""
    
    def __init__(self, config: NexusCodeConfig):

    async def execute_creative_workflow(self, workflow_type: str, parameters: Dict[str, Any]) -> Dict[str, Any]:
        """Execute predefined creative workflows"""
        
        if workflow_type == "brand_identity":
            return await self._brand_identity_workflow(parameters)
        elif workflow_type == "product_showcase":
            return await self._product_showcase_workflow(parameters)
        elif workflow_type == "concept_exploration":
            return await self._concept_exploration_workflow(parameters)
        elif workflow_type == "marketing_campaign":
            return await self._marketing_campaign_workflow(parameters)
        else:
            raise ValueError(f"Unknown workflow type: {workflow_type}")
    
    async def _brand_identity_workflow(self, params: Dict[str, Any]) -> Dict[str, Any]:
        """Complete brand identity generation workflow"""
        
        brand_name = params.get("brand_name", "Revolutionary Brand")
        industry = params.get("industry", "technology")
        values = params.get("values", ["innovation", "quality", "trust"])

        # Generate logo concepts
        logo_requests = []

                model=ImageModel.DALLE_3,

                width=512,
                height=512,
                quality=ImageQuality.ULTRA
            ))
        
        logo_batch = await self.image_generator.generate_batch(logo_requests)
        
        # Generate color palette visualizations
        palette_request = ImageGenerationRequest(
            prompt=f"color palette visualization for {brand_name} brand, modern, professional, {industry}",
            model=ImageModel.FLUX_DEV,
            style=ImageStyle.MINIMALIST,
            width=1024,
            height=512
        )
        
        palette_result = await self.image_generator.generate_image(palette_request)
        
        # Generate business card mockup
        business_card_request = ImageGenerationRequest(
            prompt=f"business card design for {brand_name}, professional, modern layout, {industry} industry",
            model=ImageModel.DALLE_3,
            style=ImageStyle.PROFESSIONAL,
            width=1024,
            height=640
        )
        
        business_card_result = await self.image_generator.generate_image(business_card_request)
        
        return {
            "workflow_type": "brand_identity",
            "brand_name": brand_name,
            "logo_concepts": [result.images[0] for result in logo_batch.results],
            "color_palette": palette_result.images[0],

                "total_time": sum(r.generation_time for r in logo_batch.results) + palette_result.generation_time + business_card_result.generation_time,
                "success_rate": len(logo_batch.results) / len(logo_requests) * 100
            }
        }

    async def _concept_exploration_workflow(self, params: Dict[str, Any]) -> Dict[str, Any]:

        logger.info(f"🎨 Exploring concept: {concept}")

                "total_time": sum(r.generation_time for r in concept_batch.results) + mood_board_result.generation_time
            }
        }


# Utility functions for integration

def save_image_from_base64(base64_data: str, filepath: Path) -> None:
    """Save base64 encoded image to file"""
    try:
        image_data = base64.b64decode(base64_data)
        with open(filepath, 'wb') as f:
            f.write(image_data)
        logger.info(f"💾 Image saved to {filepath}")
    except Exception as e:
        logger.error(f"Failed to save image: {e}")
        raise


def load_image_to_base64(filepath: Path) -> str:
    """Load image file and convert to base64"""
    try:
        with open(filepath, 'rb') as f:
            image_data = f.read()
            return base64.b64encode(image_data).decode('utf-8')
    except Exception as e:
        logger.error(f"Failed to load image: {e}")
        raise
